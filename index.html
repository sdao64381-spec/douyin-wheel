function spin(){ if(spinning){ return; } spinning = true; document.getElementById("spinButton").disabled = true; document.getElementById("result").innerHTML = "✨ 转盘进行中..."; /* 随机决定最终结果 */ const resultIndex = Math.floor( Math.random() * numbers.length ); const resultNumber = numbers[resultIndex]; /* 每一格的角度 */ const part = Math.PI * 2 / numbers.length; /* 当前角度归一化到 0 ~ 2π */ const currentAngle = ((angle % (Math.PI * 2)) + Math.PI * 2) % (Math.PI * 2); /* 目标数字需要到达正上方 */ const targetAngle = -Math.PI / 2 - ( resultIndex * part + part / 2 ); /* 计算这一次需要额外转多少角度 保证： 1. 一定向前转 2. 不会出现负数 3. 每次至少完整转 6 圈 */ let delta = targetAngle - currentAngle; delta = ((delta % (Math.PI * 2)) + Math.PI * 2) % (Math.PI * 2); /* 至少旋转 6 圈 */ const totalRotation = delta + Math.PI * 2 * 6; const startAngle = angle; const finalAngle = startAngle + totalRotation; const distance = finalAngle - startAngle; /* 每次固定 5 秒 无论第几次转， 都保持一样的速度和圈数 */ const duration = 5000; const startTime = performance.now(); function animate(now){ const elapsed = now - startTime; let progress = Math.min( elapsed / duration, 1 ); /* 缓出效果 */ const eased = 1 - Math.pow( 1 - progress, 4 ); angle = startAngle + distance * eased; draw(); if(progress < 1){ requestAnimationFrame( animate ); }else{ angle = finalAngle; draw(); spinning = false; document.getElementById( "spinButton" ).disabled = false; document.getElementById( "result" ).innerHTML = "🎉 幸运数字：" + resultNumber; /* 保存： 用户ID + 幸运数字 */ saveResult( resultNumber ); } } requestAnimationFrame( animate ); }
